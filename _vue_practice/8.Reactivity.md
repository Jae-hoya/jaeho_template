# 반응형 객체 선택

ref와 reactive?

템플린 문법이 가지고 있는 속성이기 때문에 state안에 count가 증가한다고 설정. depth가 깊어도 문제없다. 
```
<<template>
    <div>
        <button v-on:click="increment">click {{ state.count }}</button>
        <button v-on:click="increment">Deep click {{ state.deep.count }}</button>
    </div>
</template>

<script>
import { reactive } from 'vue';

export default {
    setup () {
        const state = reactive({
            count: 0,
            deep: {
                count: 0,
            }
        })
        const increment = () => {
            state.count++,
            state.deep.count++;
        }
        return {
            state,
            increment
        }
    }
}
</script>
```
